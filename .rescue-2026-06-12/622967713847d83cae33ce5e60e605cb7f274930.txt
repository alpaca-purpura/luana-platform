  function filterBiblio(q) {
    const ql = q.toLowerCase().trim();
    const results = document.getElementById('biblio-results');
    const items = results.querySelectorAll('.biblio-result');
    let visible = 0;
    items.forEach(item => {
      const nombre = item.querySelector('.br-name').textContent.toLowerCase();
      const meta = item.querySelector('.br-meta') ? item.querySelector('.br-meta').textContent.toLowerCase() : '';
      const match = !ql || nombre.includes(ql) || meta.includes(ql);
      item.style.display = match ? '' : 'none';
      if (match) visible++;
    });
    const footer = document.querySelector('.biblio-footer .hint');
    if (footer) footer.textContent = `Mostrando ${visible} de ${items.length} servicios en la biblioteca`;
    // mostrar empty state si no hay resultados
    let emptyEl = results.querySelector('.biblio-empty');
    if (!emptyEl) {
      emptyEl = document.createElement('div');
      emptyEl.className = 'biblio-empty';
      emptyEl.textContent = 'No hay resultados para "' + q + '". Prueba con otro nombre o sinónimo.';
      results.appendChild(emptyEl);
    }
    emptyEl.style.display = visible === 0 ? '' : 'none';
  }

  function selectBiblio(el, nombre, cat, peldano) {
    document.querySelectorAll('.biblio-result').forEach(r => r.removeAttribute('data-selected'));
    el.setAttribute('data-selected', 'true');
  }

  function useBiblio(nombre, peldano) {
    // Cerrar picker
    closeBiblio();
    // Pre-llenar campos con field-sparkle badges (RN-25)
    const fNombre = document.getElementById('f-nombre');
    if (fNombre) { fNombre.value = nombre; markSparkle(fNombre); }
    const fDesc = document.getElementById('f-desc');
    if (fDesc) { fDesc.value = 'Descripción oficial de "' + nombre + '" desde la biblioteca estándar. Revisa y personaliza con tu voz.'; markSparkle(fDesc); }
    const fIncluye = document.getElementById('f-incluye');
    if (fIncluye) { fIncluye.value = 'Consulta inicial · procedimiento · seguimiento'; markSparkle(fIncluye); }
    // Seleccionar peldaño matching
    document.querySelectorAll('.rung-opt').forEach(o => {
      o.removeAttribute('data-sel');
      if (o.textContent.includes(peldano.split(' ')[0])) o.setAttribute('data-sel','true');
    });
    // Actualizar breadcrumb con nombre elegido
    const entityId = document.querySelector('.entity-subnav [style*="muted-foreground"]');
    if (entityId) { entityId.innerHTML = '📚 ' + nombre + ' <span class="chip" style="margin-left:.25rem;">desde biblioteca · borrador</span>'; }
    // Activar FloatingAutosave
    showAutosave('saved');
  }

  function closeBiblio() {
    const overlay = document.getElementById('biblio-overlay');
    if (overlay) overlay.style.display = 'none';
    // Mostrar FloatingAutosave al cerrar
    const fa = document.getElementById('floating-autosave');
    if (fa) fa.style.display = '';
  }

  function markSparkle(field) {
    // Agregar badge sparkle junto al label del campo
    const label = field.closest('.form-group')?.querySelector('label') ||
                  field.previousElementSibling;
    if (label && !label.querySelector('.field-sparkle')) {
      const sp = document.createElement('span');
      sp.className = 'field-sparkle';
      sp.title = 'Pre-llenado desde la biblioteca — revísalo';
      sp.textContent = '✨ biblioteca';
      label.appendChild(sp);
    }
  }

  // FloatingAutosave: simular estados al editar campos
  let autosaveTimer;
  function showAutosave(state) {
    const fa = document.getElementById('floating-autosave');
    if (!fa) return;
    fa.style.display = '';
    if (state === 'saving') { fa.setAttribute('data-state','saving'); fa.innerHTML = '<span class="fa-dot"></span> Guardando…'; }
    else { fa.setAttribute('data-state','saved'); fa.innerHTML = '<span class="fa-dot"></span> Guardado'; }
  }
  document.addEventListener('input', e => {
    if (e.target.closest('#biblio-overlay')) return; // no disparar dentro del picker
    showAutosave('saving');
    clearTimeout(autosaveTimer);
    autosaveTimer = setTimeout(() => showAutosave('saved'), 900);
  });

  // Cerrar picker con Esc
  document.addEventListener('keydown', e => {
    if (e.key === 'Escape') closeBiblio();
  });

  // Focus automático en el input del picker al abrir
  window.addEventListener('DOMContentLoaded', () => {
    const inp = document.getElementById('biblio-input');
    if (inp) setTimeout(() => inp.focus(), 100);
  });
